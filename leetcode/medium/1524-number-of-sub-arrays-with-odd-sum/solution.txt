class Solution {
public:
    int numOfSubarrays(vector<int>& arr) {
        int prefixSum=0;
        vector<int>prefix;
        prefix.push_back(0);
        long long count=0;
        for(int i=0;i<arr.size();i++){
            prefixSum+=arr[i];
            prefix.push_back(prefixSum);
        }
        long evencount=1;
        long oddcount=0;
        for(int i=1;i<prefix.size();i++){
            if(prefix[i]%2 !=0){
                count+=evencount;
                oddcount++;
            }
               if(prefix[i]%2 ==0){
                count+=oddcount;
                evencount++;
            }
            
   
    


        }

    return count%1000000007;

        
    }
};
