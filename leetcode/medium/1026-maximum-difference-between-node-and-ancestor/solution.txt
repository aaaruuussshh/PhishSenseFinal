/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int solve(TreeNode* root, int minSoFar, int maxSoFar) {
        if(root == nullptr) {
            return 0;
        }

        minSoFar = min(minSoFar, root->val);
        maxSoFar = max(maxSoFar, root->val);

        int left = solve(root->left, minSoFar, maxSoFar);
        int right = solve(root->right, minSoFar, maxSoFar);

        int currentDiff = maxSoFar - minSoFar;

        return max(currentDiff, max(left, right));
    }
