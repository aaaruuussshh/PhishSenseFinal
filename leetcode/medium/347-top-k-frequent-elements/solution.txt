vector<int>& nums, int k) {
        unordered_map<int,int>mp;
        vector<pair<int,int>>ans;
        vector<int>final;
        for(int i=0;i<nums.size();i++){
            mp[nums[i]]++;
        }
        for(auto x : mp){
            ans.push_back({x.second,x.first});

        }
        sort(ans.rbegin(),ans.rend());
        for(int i=0;i<k;i++){
            final.push_back(ans[i].second);

        }

            
                
            
        return final;
        
        
    }
};
