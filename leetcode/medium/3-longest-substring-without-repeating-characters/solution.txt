class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_map<char,int>mp;
        int count=0;
        int maxLen=0;
        int left=0;
        for(int right=0;right<s.size();right++){
            count++;
            mp[s[right]]++;
            while(count !=mp.size()){
                mp[s[left]]--;
                count--;
                
                if(mp[s[left]] == 0){
                    mp.erase(s[left]);
                }
                left++;
                

            }
            if(count==mp.size()){
            maxLen=max(maxLen,right-left+1);
            }

          


        }
        return maxLen;
        
    }
};
