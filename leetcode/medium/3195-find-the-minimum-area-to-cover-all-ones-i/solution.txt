class Solution {
public:
    int minimumArea(vector<vector<int>>& grid) {
        int minrow= INT_MAX;
        int maxrow=-1;
        int mincol=INT_MAX;
        int maxcol=-1;
        for(int i=0;i<grid.size();i++){
            for(int j=0;j<grid[0].size();j++){
                if(grid[i][j] == 1){
                    minrow=min(minrow,i);
                    maxrow=max(maxrow,i);
                    mincol=min(mincol,j);
                    maxcol=max(maxcol,j);
                }

            }
        }
        int height = maxrow - minrow + 1;
int width = maxcol - mincol + 1;

return height * width;
    }
};
