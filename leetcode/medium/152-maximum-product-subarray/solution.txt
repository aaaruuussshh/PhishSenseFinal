 i = 0; i < nums.size(); i++) {
            runningProd *= nums[i];
            MaxProd = max(MaxProd, runningProd);

            if(runningProd == 0) {
                runningProd = 1;
            }
        }

        runningProd = 1;

        for(int i = nums.size() - 1; i >= 0; i--) {
            runningProd *= nums[i];
            MaxProd = max(MaxProd, runningProd);

            if(runningProd == 0) {
                runningProd = 1;
            }
        }

        return MaxProd;
    }
