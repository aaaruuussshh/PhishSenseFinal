class Solution {
public:
    int findMaxLength(vector<int>& nums) {
        vector<int> prefix(nums.size());
        int sum = 0;
        unordered_map<int,int>mp;
        mp[0]=-1;

    for(int i = 0; i < nums.size(); i++){
        if(nums[i] == 0){
        sum--;
    }
    else{
        sum++;
    }
    prefix[i] = sum;
}
       int maxLen=0; 
        for(int i = 0; i < prefix.size(); i++){
    if(mp.find(prefix[i]) != mp.end()){
        maxLen=max(maxLen,i-mp[prefix[i]]);
    }
    else{
        mp[prefix[i]] = i;
    }
}
return maxLen;
    }
};
