 root, int& nodes){
        if(root==nullptr){
            return {0,0};
        }

        pair<int,int>  left =solve(root->left,nodes);
        pair<int,int> right =solve(root->right,nodes);
       

        int sum=left.first+right.first+root->val;
         int count = 1 + left.second + right.second;
        
        if((sum/count)==root->val){
            nodes++;

        }
        return {sum, count};

    }
    int averageOfSubtree(TreeNode* root) {
       int nodes=0;
        solve(root,nodes);
        return nodes;
        
    }
};
