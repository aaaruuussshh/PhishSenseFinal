    int maximumSum(vector<int>& nums) {
        unordered_map<int, int> mp;
        int maxSum=-1;
        for(int i=0;i<nums.size();i++){
            int sum=DigitSum(nums[i]);
            if(mp.find(sum) !=mp.end()){
                maxSum=max(maxSum,nums[i] + mp[sum]);
                mp[sum] = max(mp[sum],nums[i]);
            }
            else{
                mp[sum]=nums[i];
            }

        }

                
            

            
        
        return maxSum;
        
    }
