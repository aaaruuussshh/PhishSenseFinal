     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    pair<int,TreeNode* > solve(TreeNode* root){
        if(root==nullptr){
            return {0,0};
        }
        pair<int, TreeNode*>  left=solve(root->left);
        pair<int, TreeNode*> right = solve(root->right);
        TreeNode* BestSubRoot;
        if(left.first>right.first){
            BestSubRoot=left.second;
        }
        else if(right.first > left.first) {
        BestSubRoot = right.second;
}
    else {
        BestSubRoot = root;
}
return {1 + max(left.first, right.first), BestSubRoot};
    }
    TreeNode* subtreeWithAllDeepest(TreeNode* root) {
        pair<int, TreeNode*> result = solve(root);
        return result.second;

        
    }
};
