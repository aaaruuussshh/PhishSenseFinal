class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
        unordered_map<int,int>mp;
        vector<int> prefix;
        int currSum=0;
        int count=0;
        mp[0]=1;
        for(int i=0; i<nums.size();i++){
            currSum+=nums[i];
            prefix.push_back(currSum);

        }
        for(int i=0;i<prefix.size();i++){
            int need=prefix[i]-goal;
            if(mp.find(need) != mp.end()){
                count+=mp[need];
            }
            mp[prefix[i]]++;
        }
        return count;
        
    }
};
