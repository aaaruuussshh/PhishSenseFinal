class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int currSum=0;
        int maxSum=INT_MIN;
        int i=0;
        while(i<nums.size()){
            currSum+=nums[i];
            if(nums.size() ==1 && nums[i] <0){
                return nums[i];
            }
maxSum=max(maxSum,currSum);
            if(currSum<0){
                currSum=0;

            }
            
            i++;
            
        }
        return maxSum;

        
    }
};
