k>=nums.size()){
                ans.push_back(-1);
            }
            else{
                if(i==k){
                    for(int j=i-k;j<=i+k;j++){
                        TempSum+=nums[j];
                    }
                }
                else{
                    TempSum-=nums[i-k-1];
                    TempSum+=nums[i+k];
                }

                int TempAvg=TempSum/(2*k+1);
                ans.push_back(TempAvg);
            }
        }

        return ans;
    }
