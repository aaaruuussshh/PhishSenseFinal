class Solution {
public:
    int numPairsDivisibleBy60(vector<int>& time) {
        unordered_map<int,int>mp;
        int cnt=0;
        for(int i=0;i<time.size();i++){
            int rem=time[i]%60;
            
            int req=(60-rem)%60;
            if(mp.find(req) !=mp.end()){
                cnt+=mp[req];
                
            }
            mp[rem]++;


        }
        return cnt;
        
    }
};
