class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
        int left=0;
        int ZeroCount=0;
        int MaxLength=0;
        for(int right=0;right<nums.size();right++){
             if(nums[right]==0){
                ZeroCount++;

             }
             while(ZeroCount>k){
                if(nums[left]==0){
                    ZeroCount--;
                }
                left++;
             }
             MaxLength=max(MaxLength,right-left+1);


        }
        return MaxLength;
        
    }
};
