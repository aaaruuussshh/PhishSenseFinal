de* dfs(TreeNode* root, TreeNode* p, TreeNode* q){
        if(root==nullptr){
            return NULL;
        }
        if(root ==p || root == q){
           return root; 
        }
        TreeNode* left=dfs(root->left,p,q);
        TreeNode* right=dfs(root->right,p,q);
        if(left !=nullptr && right!=nullptr){
            return root;
        }
        if(left!=nullptr){
            return left;
        }
        if(right!=nullptr){
            return right;
        }

     return nullptr;
            

        


    }
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
