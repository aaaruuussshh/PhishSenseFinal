class Solution {
public:
    vector<int> successfulPairs(vector<int>& spells, vector<int>& potions, long long success) {
        vector<int> ans;

        sort(potions.begin(), potions.end());

        int n = potions.size();

        for (int i = 0; i < spells.size(); i++) {
            int l = 0;
            int r = n - 1;
            int idx = n;

            while (l <= r) {
                int mid = l + (r - l) / 2;

                if (1LL * spells[i] * potions[mid] >= success) {
                    idx = mid;
                    r = mid - 1;
                }
                else {
                    l = mid + 1;
                }
            }

            ans.push_back(n - idx);
        }

        return ans;
    }
};
