class Solution {
public:
    long long countCompleteDayPairs(vector<int>& hours) {
        unordered_map<int,int>mp;
        long long cnt=0;
        for(int i=0;i<hours.size();i++){
            int rem=hours[i]%24;
            int req=(24-rem)%24;
            if(mp.find(req) != mp.end()){
                cnt+=mp[req];
            }
            mp[rem]++;
            
        }
        return cnt;
        
    }
};
