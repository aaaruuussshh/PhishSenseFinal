);
        int right=solve(root->right);
        int sum=left+right+root->val;
        mp[sum]++;
        return left+right+root->val;

    }
    vector<int> findFrequentTreeSum(TreeNode* root) {
        solve(root);
        int maxFreq=0;
        vector<int>ans;
        for(auto [sum,freq] : mp){
            if(freq>maxFreq){
                maxFreq=freq;
                ans.clear();
                ans.push_back(sum);

            }
            else if(freq==maxFreq){
                ans.push_back(sum);
            }


        }

        return ans;
    }
