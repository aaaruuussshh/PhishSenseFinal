> findFrequentTreeSum(TreeNode* root) {
        vector<int> temp;
        int sum=0;
        solve(root,temp,sum);
        
        unordered_map<int,int> mp;
        for(int i=0;i<temp.size();i++){
            mp[temp[i]]++;
        }
        int maxi=INT_MIN;
        for(auto x:mp){
            maxi = max(maxi,x.second);


        }
        vector<int>ans;
        for(auto x:mp){
            if(x.second==maxi){
                ans.push_back(x.first);
            }
        }
        return ans;

        
        
    }
};
