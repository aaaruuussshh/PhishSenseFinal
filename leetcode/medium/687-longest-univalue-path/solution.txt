class Solution {
public:

    int solve(TreeNode* root, int prevValue, int length, int& maxLength) {
        if (root == nullptr) {
            return length;
        }

        if (root->val == prevValue) {
            length++;
            maxLength = max(maxLength, length);
        }
        else {
            return length;
        }

        int left = solve(root->left, root->val, length, maxLength);
        int right = solve(root->right, root->val, length, maxLength);

        return max(left, right);
    }
