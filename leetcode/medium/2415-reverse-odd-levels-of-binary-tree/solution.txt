    TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    TreeNode* reverseOddLevels(TreeNode* root) {
        vector<int>ans;
        queue<TreeNode*>q;
        q.push(root);
        bool odd=false;
        while(!q.empty()){
            int size=q.size();
            vector<TreeNode*>tp;
            for(int i=0;i<size;i++){
                TreeNode* temp=q.front();
                q.pop();
                tp.push_back(temp);
                if(temp->left!=nullptr){
                    q.push(temp->left);
                }
                if(temp->right !=nullptr){
                    q.push(temp->right);
                }


            }
            if (odd) {
    int a = 0;
    int b = tp.size() - 1;

    while (a < b) {
        swap(tp[a]->val, tp[b]->val);
        a++;
        b--;
    }
}

odd = !odd;
        }

     return root;   
    }
};
