class Solution {
public:
    int equalPairs(vector<vector<int>>& grid) {
    int rows=grid.size();
    int count=0;
    int cols=grid[0].size();
    
    
    for(int r=0;r<rows;r++){
        for(int c=0;c<cols;c++){
            bool check=true;
            for(int k=0;k<rows;k++){
                if(grid[r][k] != grid[k][c]){
                    check=false;
                    break;
                }
                
                check=true;
                
            }


            if(check){
                count++;
            }
        
        }
    }
    
      return count;  
    }
};
