                }

                for(int d = 0; d < 8; d++) {
                    int nr = r + dr[d];
                    int nc = c + dc[d];


                    if(nr < 0 || nc < 0 || nr >= n || nc >= n) {
                        continue;
                    }

                    }

            }

        }
