   vector<vector<int>> solve(TreeNode* root, int targetSum, int sum, vector<vector<int>>& ans, vector<int>& path){
        if(root==nullptr){
            return ans;
        }
        path.push_back(root->val);
        sum+=root->val;
        if(sum==targetSum && root->left==nullptr && root->right==nullptr){
            ans.push_back(path);
            
        }
        
        solve(root->left,targetSum,sum,ans,path);
        solve(root->right,targetSum,sum,ans,path);
        path.pop_back();
        return ans;


    }
    vector<vector<int>> pathSum(TreeNode* root, int targetSum) {
        vector<vector<int>> ans;
        vector<int> path;
        int sum=0;
        return solve(root,targetSum,sum,ans,path);

        
    }
