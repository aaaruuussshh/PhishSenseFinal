
        vector<int>ans;
        if(root==nullptr){
            return ans ;
        }
        
       queue<TreeNode*>q;
        q.push(root);
        
        while(!q.empty()){
            int level=q.size();
            int maxVal=INT_MIN;
            for(int i=0;i<level;i++){
                TreeNode* temp=q.front();
            q.pop();
            maxVal = max(maxVal, temp->val);
            
            if(temp->left){
                q.push(temp->left);
            }
            if(temp->right){
                q.push(temp->right);
            }
            

            }
            ans.push_back(maxVal);
            
            

            

        }
        return ans;
        
    }
