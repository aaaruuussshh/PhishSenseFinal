class Solution {
public:
    long long zeroFilledSubarray(vector<int>& nums) {
        long long count=0;
        int l=0;
        int r=0;
while (r < nums.size()) {
    if (nums[r] == 0) {
        count += r - l + 1;
    }
    else {
        l = r + 1;
    }

    r++;
}
return count;
        
    }
};
