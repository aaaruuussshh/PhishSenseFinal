int count = 0;

        for (int i = 0; i < A.size(); i++) {

            if (A[i] == B[i]) {
                count++;
            }
            else {
                if (Bset.find(A[i]) != Bset.end()) {
                    count++;
                }

                if (Aset.find(B[i]) != Aset.end()) {
                    count++;
                }
            }

            Aset.insert(A[i]);
            Bset.insert(B[i]);

            C.push_back(count);
        }

        return C;
    }
