class Solution {
public:
    vector<int> findThePrefixCommonArray(vector<int>& A, vector<int>& B) {
       unordered_set<int>Aset;
       unordered_set<int>Bset;
       vector<int>C;
       
       for(int i=0;i<A.size();i++){
        int count=0;
        Aset.insert(A[i]);
        Bset.insert(B[i]);
        for (int x : Aset){
            if (Bset.find(x) != Bset.end()) {
    count++;
}
       

        }
         C.push_back(count);
        
       } 
       return C;
    }
};
