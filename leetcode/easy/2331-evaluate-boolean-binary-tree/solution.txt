{
public:
    bool dfs(TreeNode* root){
                if(root==nullptr){
            return false;
        }
        bool left = dfs(root->left);
        bool right=dfs(root->right);
        if(root->left == nullptr && root->right==nullptr){
            if(root->val==0){
                return false;
            }
            else{
                return true;
            }

        }
        if(root->val==2){
            return left || right;
        }
        else{
            return left && right;

        }
        return false;

        

    }
    bool evaluateTree(TreeNode* root) {
        return dfs(root);


        
    }
};
