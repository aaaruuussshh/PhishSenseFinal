class Solution {
public:
vector<double>avg;
    double minimumAverage(vector<int>& nums) {
        
        sort(nums.begin(),nums.end());
         int i=0;
        int j=nums.size()-1;
        while(i<j){
       
        double average = (nums[j] + nums[i])/2.0;
        avg.push_back(average);
        i++;
        j--;
        }

        
    
   double minElement=*min_element(avg.begin(),avg.end());
   return minElement;
    }
};
