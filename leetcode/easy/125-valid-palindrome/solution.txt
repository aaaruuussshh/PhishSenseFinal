class Solution {
public:
    bool isPalindrome(string s) {
        string final="";
        for(int i=0;i<s.size();i++){
            if(isalnum(s[i])){
                final.push_back(tolower(s[i]));

            }
            
        }
        for(int i=0;i<final.size()/2;i++){
            if(final[i]!=final[final.size()-i-1]){
                return false;
            }
        }
        return true;

        
        
    }
};
