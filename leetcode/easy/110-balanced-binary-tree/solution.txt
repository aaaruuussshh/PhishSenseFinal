class Solution {
public:
    int solve(TreeNode* root) {
        if(root == nullptr) {
            return 0;
        }

        int left = solve(root->left);
        int right = solve(root->right);

        if(left == -1 || right == -1) {
            return -1;
        }

        int balance = abs(left - right);

        if(balance <= 1) {
            return 1 + max(left, right);
        }

        return -1;
    }

    bool isBalanced(TreeNode* root) {
        return solve(root) != -1;
    }
};
