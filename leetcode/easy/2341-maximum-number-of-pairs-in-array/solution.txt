class Solution {
public:
    vector<int> numberOfPairs(vector<int>& nums) {
        int pairCount=0;
        int leftCount=0;
        vector<int>ans;
        unordered_map<int,int>mp;
        for(int i=0;i<nums.size();i++){
            mp[nums[i]]++;
        }
        for(auto x : mp){
            if(x.second%2==0){
                pairCount+=x.second/2;
            }
            else{
                if(x.second>2){
                    pairCount+=x.second/2;
                    leftCount+=1;
                }
                else if(x.second ==1){
                    leftCount+=1;
                }


            }
        }
        ans.push_back(pairCount);
        ans.push_back(leftCount);
        return ans;


        
    }
};
