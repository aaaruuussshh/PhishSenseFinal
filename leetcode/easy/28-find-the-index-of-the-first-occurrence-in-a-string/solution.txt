class Solution {
public:
    int strStr(string haystack, string needle) {
        int count=0;
                int n = haystack.size();
        int m = needle.size();
        for(int i=0;i<=n-m;i++){
            if( haystack.substr(i,needle.size()) == needle){
                return count;
            }
            count++;
        }
        return -1;

    }
};
