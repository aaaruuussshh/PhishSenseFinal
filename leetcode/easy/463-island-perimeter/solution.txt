    int islandPerimeter(vector<vector<int>>& grid) {
                int m = grid.size();
        int n = grid[0].size();

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1)
                    return dfs(grid, i, j);
            }
        }

        return 0;
