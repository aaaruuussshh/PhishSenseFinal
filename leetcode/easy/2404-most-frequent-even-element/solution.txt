int mostFrequentEven(vector<int>& nums) {
        int MostFreq = 0;
        int ans = -1;
        map<int, int> mp;
        for (int i = 0; i < nums.size(); i++) {
            mp[nums[i]]++;
        }
        for (auto x : mp) {
            if (x.first % 2 == 0) {
                if (x.second > MostFreq) {
                MostFreq = x.second;
                ans = x.first;
                }
                else if (x.second == MostFreq) {
                    ans = min(ans, x.first);
             }
            }
        }

        return ans;
    }
};
