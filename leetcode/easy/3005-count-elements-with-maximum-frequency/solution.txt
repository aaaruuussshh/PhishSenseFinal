class Solution {
public:
    int maxFrequencyElements(vector<int>& nums) {
        int count=0;
        unordered_map<int,int>mp;
        for(int i=0;i<nums.size();i++){
            mp[nums[i]]++;
        }
        int maxFreq=INT_MIN;
        for(auto x:mp){
            maxFreq=max(maxFreq,x.second);
        }
        for(auto x:mp){
            if(x.second==maxFreq){
                count+=x.second;
            }
        }
        return count;
    }
};
