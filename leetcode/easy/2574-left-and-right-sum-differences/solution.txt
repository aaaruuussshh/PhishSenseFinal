class Solution {
public:
    vector<int> leftRightDifference(vector<int>& nums) {
        int leftSum=0;
        int rightSum=0;
        vector<int>leftarr;
        vector<int>rightarr;
        vector<int>ans;
        leftarr.push_back(0);
        for(int i=0;i<nums.size()-1;i++){
            leftSum+=nums[i];
            leftarr.push_back(leftSum);
            
        }
        
                for(int i=0;i<nums.size()-1;i++){
                    rightSum=0;
             for(int j=i+1;j<nums.size();j++){       
            rightSum+=nums[j];
           
            }
             rightarr.push_back(rightSum);
            
        }
        rightarr.push_back(0);

        for(int i=0;i<rightarr.size();i++){
            ans.push_back(abs(leftarr[i]-rightarr[i]));


        }
return ans;

        
    }
};
