i++) {
            digit += to_string(digits[i]);
        }

        // Add 1 directly to the string
        int i = digit.size() - 1;

        while (i >= 0 && digit[i] == '9') {
            digit[i] = '0';
            i--;
        }

        if (i >= 0) {
            digit[i]++;
        } else {
            digit = "1" + digit;
        }

        vector<int> ans;

        for (int i = 0; i < digit.size(); i++) {
            ans.push_back(digit[i] - '0');
        }

        return ans;
    }
