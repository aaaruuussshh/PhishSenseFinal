class Solution {
public:
    long long countSubarrays(vector<int>& nums, long long k) {
        int left=0;
        long long currProdSum=0;
        long long currSum=0;
        long long count=0;
        for(int right=0;right<nums.size();right++){
            currSum+=nums[right];
            currProdSum=currSum*(right-left+1);

            while(currProdSum>=k){
                currSum-=nums[left];
                  left++;

                currProdSum=currSum*(right-left+1);
              
            }
            count+=right-left+1LL;

          


            
            

        }
        return count;
        
    }
};
